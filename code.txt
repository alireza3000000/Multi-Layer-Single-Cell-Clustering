import numpy as np
import scipy.sparse as sp
import os
import torch
import torch.nn as nn
import torch.optim as optim
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score, calinski_harabasz_score, davies_bouldin_score
from sklearn.decomposition import TruncatedSVD
import gc

# ===== مسیر Google Drive (درست) =====
paper18 = "/content/drive/MyDrive/natayej"

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"Device: {device}")

# ===== بارگذاری =====
print("📂 بارگذاری...")
rna_norm = sp.load_npz(os.path.join(paper18, "rna_norm.npz"))
protein_norm = sp.load_npz(os.path.join(paper18, "protein_norm.npz"))
glycolysis = np.load(os.path.join(paper18, "glycolysis_score.npy"))
oxphos = np.load(os.path.join(paper18, "oxphos_score.npy"))

print(f"RNA: {rna_norm.shape}")
print(f"پروتئین: {protein_norm.shape}")
print(f"گلیکولیز: {glycolysis.shape}")
print(f"OXPHOS: {oxphos.shape}")

# ===== بردار امضا =====
metabolism = sp.csr_matrix(np.column_stack([glycolysis, oxphos]))
signature = sp.hstack([rna_norm, protein_norm, metabolism], format='csr')
print(f"بردار امضا: {signature.shape}")

# ===== کاهش بُعد =====
svd = TruncatedSVD(n_components=50, random_state=42)
X = svd.fit_transform(signature)
print(f"PCA: {X.shape}")

# ===== Autoencoder =====
class Autoencoder(nn.Module):
    def __init__(self, input_dim, latent_dim=10):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, 500),
            nn.ReLU(),
            nn.Linear(500, 500),
            nn.ReLU(),
            nn.Linear(500, 2000),
            nn.ReLU(),
            nn.Linear(2000, latent_dim)
        )
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, 2000),
            nn.ReLU(),
            nn.Linear(2000, 500),
            nn.ReLU(),
            nn.Linear(500, 500),
            nn.ReLU(),
            nn.Linear(500, input_dim)
        )

    def forward(self, x):
        z = self.encoder(x)
        x_recon = self.decoder(z)
        return z, x_recon

# ===== آموزش =====
input_dim = X.shape[1]
latent_dim = 10
ae = Autoencoder(input_dim, latent_dim).to(device)
optimizer = optim.Adam(ae.parameters(), lr=1e-3)
criterion = nn.MSELoss()

X_tensor = torch.FloatTensor(X).to(device)
batch_size = 256
n_epochs = 50

print("🔄 آموزش Autoencoder...")
for epoch in range(n_epochs):
    ae.train()
    total_loss = 0
    indices = np.random.permutation(len(X_tensor))

    for i in range(0, len(X_tensor), batch_size):
        batch_idx = indices[i:i+batch_size]
        batch = X_tensor[batch_idx]

        optimizer.zero_grad()
        z, x_recon = ae(batch)
        loss = criterion(x_recon, batch)
        loss.backward()
        optimizer.step()

        total_loss += loss.item()

    if (epoch + 1) % 10 == 0:
        print(f"Epoch {epoch+1}/{n_epochs}, Loss: {total_loss:.4f}")

# ===== استخراج ویژگی‌ها =====
ae.eval()
with torch.no_grad():
    Z, _ = ae(X_tensor)
    Z = Z.cpu().numpy()

print(f"Latent: {Z.shape}")

# ===== خوشه‌بندی =====
n_clusters = 28
kmeans = KMeans(n_clusters=n_clusters, random_state=42, n_init=10)
labels = kmeans.fit_predict(Z)

# ===== معیارها =====
sil = silhouette_score(Z, labels, sample_size=2000, random_state=42)
ch = calinski_harabasz_score(Z, labels)
db = davies_bouldin_score(Z, labels)

print(f"\n📊 نتایج DEC (شبکه عصبی):")
print(f"  - خوشه‌ها: {n_clusters}")
print(f"  - Silhouette: {sil:.4f}")
print(f"  - Calinski-Harabasz: {ch:.2f}")
print(f"  - Davies-Bouldin: {db:.4f}")

# ===== ذخیره =====
np.save(os.path.join(paper18, "dec_labels.npy"), labels)
np.save(os.path.join(paper18, "dec_latent.npy"), Z)
np.save(os.path.join(paper18, "dec_metrics.npy"), np.array([sil, ch, db]))
print("\n✅ ذخیره شد: dec_labels.npy, dec_latent.npy, dec_metrics.npy")
